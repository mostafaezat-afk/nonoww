export default { async fetch(request, env) { const url = new URL(request.url); // CORS const corsHeaders = makeCorsHeaders(request, env); if (request.method === 'OPTIONS') { return new Response(null, { status: 204, headers: corsHeaders }); } try { if (url.pathname === '/api/complaints' && request.method === 'GET') { const clientId = request.headers.get('x-client-id') || url.searchParams.get('client_id') || ''; const data = await listComplaints(env, clientId); return json(data, 200, corsHeaders); } if (url.pathname === '/api/complaints' && request.method === 'POST') { const body = await request.json(); await verifyTurnstile(env, body.turnstile_token, request); const created = await createComplaint(env, body); return json(created, 200, corsHeaders); } if (url.pathname === '/api/vote' && request.method === 'POST') { const body = await request.json(); const out = await castVote(env, body); return json(out, 200, corsHeaders); } if (url.pathname === '/api/comments' && request.method === 'GET') { const complaintId = url.searchParams.get('complaint_id'); if (!complaintId) return json({ error:'complaint_id مطلوب' }, 400, corsHeaders); const data = await listComments(env, complaintId); return json(data, 200, corsHeaders); } if (url.pathname === '/api/comments' && request.method === 'POST') { const body = await request.json(); const out = await addComment(env, body); return json(out, 200, corsHeaders); } return json({ error: 'Not Found' }, 404, corsHeaders); } catch (err) { return json({ error: err.message || 'Server error' }, 500, corsHeaders); } } }; function json(obj, status=200, headers={}) { return new Response(JSON.stringify(obj), { status, headers: { 'Content-Type': 'application/json; charset=utf-8', ...headers } }); } function makeCorsHeaders(request, env){ const origin = request.headers.get('Origin') || ''; const allowed = (env.ALLOWED_ORIGINS || '').split(',').map(s=>s.trim()).filter(Boolean); // لو ما حطتش ALLOWED_ORIGINS، هيسمح للجميع (ممكن تغيّره للأمان) const allowOrigin = allowed.length ? (allowed.includes(origin) ? origin : allowed[0]) : '*'; return { 'Access-Control-Allow-Origin': allowOrigin, 'Access-Control-Allow-Methods': 'GET,POST,OPTIONS', 'Access-Control-Allow-Headers': 'Content-Type,x-client-id', 'Access-Control-Max-Age': '86400' }; } /* ====================== Turnstile verify ====================== */ async function verifyTurnstile(env, token, request){ if(!token) throw new Error('Turnstile token ناقص'); const ip = request.headers.get('CF-Connecting-IP') || ''; const form = new FormData(); form.append('secret', env.TURNSTILE_SECRET); form.append('response', token); if(ip) form.append('remoteip', ip); const res = await fetch('https://challenges.cloudflare.com/turnstile/v0/siteverify', { method:'POST', body: form }); const data = await res.json(); if(!data.success){ throw new Error('فشل التحقق من Turnstile'); } } /* ====================== Supabase helpers ====================== */ async function sbFetch(env, path, options={}){ const url = env.SUPABASE_URL.replace(/\/$/,'') + path; const res = await fetch(url, { ...options, headers: { 'apikey': env.SUPABASE_SERVICE_KEY, 'Authorization': `Bearer ${env.SUPABASE_SERVICE_KEY}`, ...options.headers } }); const txt = await res.text(); const data = txt ? safeJson(txt) : null; if(!res.ok){ const msg = (data && data.message) ? data.message : `Supabase error: ${res.status}`; throw new Error(msg); } return data; } function safeJson(txt){ try{ return JSON.parse(txt); }catch{ return { raw: txt }; } } /* ====================== Storage upload images: [{name, data_url}] ====================== */ async function uploadDataUrlToStorage(env, dataUrl, filename){ if(!dataUrl || !dataUrl.startsWith('data:image/')) return null; const [meta, b64] = dataUrl.split(','); const mime = meta.match(/data:(.*?);base64/)[1] || 'image/jpeg'; const bytes = Uint8Array.from(atob(b64), c => c.charCodeAt(0)); const ext = mime.includes('png') ? 'png' : (mime.includes('webp') ? 'webp' : 'jpg'); const path = `complaints/${Date.now()}-${Math.random().toString(16).slice(2)}.${ext}`; // PUT object const putRes = await fetch( env.SUPABASE_URL.replace(/\/$/,'') + `/storage/v1/object/${env.SUPABASE_STORAGE_BUCKET}/${path}`, { method:'POST', // Supabase Storage accepts POST for upload headers:{ 'apikey': env.SUPABASE_SERVICE_KEY, 'Authorization': `Bearer ${env.SUPABASE_SERVICE_KEY}`, 'Content-Type': mime, 'x-upsert': 'true' }, body: bytes } ); if(!putRes.ok){ const t = await putRes.text(); throw new Error('فشل رفع الصورة: ' + t); } // Public URL (لو bucket public) const publicUrl = env.SUPABASE_URL.replace(/\/$/,'') + `/storage/v1/object/public/${env.SUPABASE_STORAGE_BUCKET}/${path}`; return publicUrl; } /* ====================== API: list complaints Returns counts + user_vote based on client_id ====================== */ async function listComplaints(env, clientId){ // 1) get complaints const items = await sbFetch(env, `/rest/v1/complaints?select=*&order=created_at.desc`, { method:'GET' }); // 2) counts from votes/comments via aggregated views // - we assume you created views: v_vote_counts and v_comment_counts (SQL below) const voteCounts = await sbFetch(env, `/rest/v1/v_vote_counts?select=complaint_id,support_count,oppose_count`, { method:'GET' }); const commentCounts = await sbFetch(env, `/rest/v1/v_comment_counts?select=complaint_id,comments_count`, { method:'GET' }); const vcMap = new Map(voteCounts.map(v=>[v.complaint_id, v])); const ccMap = new Map(commentCounts.map(v=>[v.complaint_id, v])); // 3) user vote for this client (optional) let myVotes = []; if(clientId){ myVotes = await sbFetch(env, `/rest/v1/votes?select=complaint_id,vote&client_id=eq.${encodeURIComponent(clientId)}`, { method:'GET' }); } const myMap = new Map(myVotes.map(v=>[v.complaint_id, v.vote])); const merged = items.map(c=>{ const vc = vcMap.get(c.id) || {}; const cc = ccMap.get(c.id) || {}; return { ...c, support_count: vc.support_count || 0, oppose_count: vc.oppose_count || 0, comments_count: cc.comments_count || 0, user_vote: myMap.get(c.id) || null }; }); return { items: merged }; } /* ====================== API: create complaint ====================== */ async function createComplaint(env, body){ const required = ['client_id','title','type','lat','lng']; for(const k of required){ if(!body[k]) throw new Error(`الحقل مطلوب: ${k}`); } // upload images if any const imageUrls = []; const images = Array.isArray(body.images) ? body.images : []; for(const img of images.slice(0,5)){ if(img?.data_url){ const url = await uploadDataUrlToStorage(env, img.data_url, img.name || 'image.jpg'); if(url) imageUrls.push(url); } } const insert = [{ client_id: String(body.client_id).slice(0,120), title: String(body.title).slice(0,120), type: String(body.type).slice(0,30), desc: (body.desc||'').toString().slice(0,2000), street: (body.street||'بشتيل').toString().slice(0,120), name: (body.name||'مجهول').toString().slice(0,80), lat: Number(body.lat), lng: Number(body.lng), status: 'open', image_urls: imageUrls }]; const created = await sbFetch(env, `/rest/v1/complaints`, { method:'POST', headers:{ 'Content-Type':'application/json', 'Prefer':'return=representation' }, body: JSON.stringify(insert) }); return { ok:true, item: created?.[0] || null }; } /* ====================== API: vote - unique (complaint_id, client_id) - toggles: same vote again => remove - switching vote => update ====================== */ async function castVote(env, body){ const { complaint_id, client_id, vote } = body || {}; if(!complaint_id || !client_id || !vote) throw new Error('بيانات التصويت ناقصة'); if(!['support','oppose'].includes(vote)) throw new Error('نوع التصويت غير صحيح'); // check existing const existing = await sbFetch(env, `/rest/v1/votes?select=id,vote&complaint_id=eq.${complaint_id}&client_id=eq.${encodeURIComponent(client_id)}`, { method:'GET' }); if(existing.length){ const row = existing[0]; if(row.vote === vote){ // toggle off => delete await sbFetch(env, `/rest/v1/votes?id=eq.${row.id}`, { method:'DELETE' }); // return counts const counts = await voteCounts(env, complaint_id); return { ok:true, ...counts, user_vote: null }; }else{ // update await sbFetch(env, `/rest/v1/votes?id=eq.${row.id}`, { method:'PATCH', headers:{'Content-Type':'application/json'}, body: JSON.stringify({ vote }) }); const counts = await voteCounts(env, complaint_id); return { ok:true, ...counts, user_vote: vote }; } } else { // insert await sbFetch(env, `/rest/v1/votes`, { method:'POST', headers:{'Content-Type':'application/json'}, body: JSON.stringify([{ complaint_id, client_id, vote }]) }); const counts = await voteCounts(env, complaint_id); return { ok:true, ...counts, user_vote: vote }; } } async function voteCounts(env, complaintId){ const vc = await sbFetch(env, `/rest/v1/v_vote_counts?select=support_count,oppose_count&complaint_id=eq.${complaintId}`, { method:'GET' }); const row = vc[0] || { support_count:0, oppose_count:0 }; return { support_count: row.support_count||0, oppose_count: row.oppose_count||0 }; } /* ====================== API: comments ====================== */ async function listComments(env, complaintId){ const items = await sbFetch(env, `/rest/v1/comments?select=id,complaint_id,name,text,image_url,created_at&complaint_id=eq.${complaintId}&order=created_at.asc`, { method:'GET' }); return { items }; } async function addComment(env, body){ const { complaint_id, client_id, name, text, image_data_url } = body || {}; if(!complaint_id || !client_id || !text) throw new Error('بيانات التعليق ناقصة'); let imageUrl = null; if(image_data_url){ imageUrl = await uploadDataUrlToStorage(env, image_data_url, 'comment.jpg'); } await sbFetch(env, `/rest/v1/comments`, { method:'POST', headers:{'Content-Type':'application/json'}, body: JSON.stringify([{ complaint_id, client_id: String(client_id).slice(0,120), name: (name||'مجهول').toString().slice(0,80), text: String(text).slice(0,2000), image_url: imageUrl }]) }); return { ok:true }; }
